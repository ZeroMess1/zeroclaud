// Zero Messenger - Firestore Security Rules
// Скопируйте этот код в Firebase Console → Firestore → Rules

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // ==================== ФУНКЦИИ ====================
    function isAuthenticated() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return request.auth.uid == uid;
    }

    function isParticipant(participants) {
      return request.auth.uid in participants;
    }

    function isSender(senderId) {
      return request.auth.uid == senderId;
    }

    // ==================== ПОЛЬЗОВАТЕЛИ ====================
    // Коллекция users - профили пользователей
    match /users/{uid} {
      // Любой авторизованный пользователь может читать профили
      allow read: if isAuthenticated();
      
      // Только хозяин может писать
      allow write: if isOwner(uid);
      
      // Подколлекции
      match /{document=**} {
        allow read, write: if isOwner(uid);
      }
    }

    // ==================== ЧАТЫ ====================
    // Коллекция chats - разговоры между пользователями
    match /chats/{chatId} {
      // Только участники могут читать
      allow read: if isAuthenticated() && isParticipant(resource.data.participants);
      
      // Только участники могут писать (создавать/обновлять чат)
      allow create: if isAuthenticated() && isParticipant(request.resource.data.participants);
      allow update: if isAuthenticated() && isParticipant(resource.data.participants);
      
      // Удаление запрещено (используем флаги deleted вместо этого)
      allow delete: if false;

      // Сообщения в чате
      match /messages/{messageId} {
        allow read: if isParticipant(get(/databases/$(database)/documents/chats/$(chatId)).data.participants);
        allow create: if isAuthenticated() && request.resource.data.senderId == request.auth.uid;
        allow update: if isSender(resource.data.senderId);
      }
    }

    // ==================== СООБЩЕНИЯ ====================
    // Коллекция messages - все сообщения
    match /messages/{messageId} {
      // Можно читать сообщения из чатов где ты участник
      allow read: if isAuthenticated() && 
        isParticipant(get(/databases/$(database)/documents/chats/$(resource.data.chatId)).data.participants);
      
      // Можно создавать сообщения (проверяется в чатах)
      allow create: if isAuthenticated() && 
        isSender(request.resource.data.senderId) &&
        isParticipant(get(/databases/$(database)/documents/chats/$(request.resource.data.chatId)).data.participants);
      
      // Только автор может обновлять (редактировать)
      allow update: if isAuthenticated() && 
        isSender(resource.data.senderId) &&
        // Можно изменять только: text, edited, editedAt, deletedForMe, deletedForAll, read
        request.resource.data.messageId == resource.data.messageId &&
        request.resource.data.senderId == resource.data.senderId &&
        request.resource.data.createdAt == resource.data.createdAt;
      
      // Только автор может удалять
      allow delete: if isAuthenticated() && isSender(resource.data.senderId);
    }

    // ==================== БЛОКИРОВКИ ====================
    // Коллекция blocks - заблокированные пользователи
    match /blocks/{blockId} {
      // Можно читать свои блокировки
      allow read: if isAuthenticated() && 
        (isOwner(resource.data.blockerId) || isOwner(resource.data.blockedId));
      
      // Можно создавать блокировку от себя
      allow create: if isAuthenticated() && 
        request.auth.uid == request.resource.data.blockerId;
      
      // Только блокирующий может удалять
      allow delete: if isAuthenticated() && 
        request.auth.uid == resource.data.blockerId;
    }

    // ==================== УВЕДОМЛЕНИЯ ====================
    // Опциональная коллекция для уведомлений
    match /notifications/{userId}/messages/{notificationId} {
      allow read: if isAuthenticated() && isOwner(userId);
      allow create: if isAuthenticated();
      allow delete: if isAuthenticated() && isOwner(userId);
    }

    // ==================== ЛОГИРОВАНИЕ ====================
    // Опциональная коллекция для логирования активности
    match /logs/{logId} {
      allow create: if isAuthenticated();
      allow read: if false; // Логи не должны быть доступны
    }

    // ==================== ЗАПРЕТ ОСТАЛЬНОГО ====================
    // По умолчанию запрещаем доступ
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

users/{uid}
├── userId: number (1000-9999)
├── name: string
├── email: string
├── avatar: string (base64)
├── about: string
├── birthday: string (YYYY-MM-DD)
├── hideAbout: boolean
├── hideBirthday: boolean
├── online: boolean
├── lastSeen: timestamp
├── createdAt: timestamp
└── updatedAt: timestamp

chats/{chatId} (format: uid1_uid2)
├── chatId: string
├── participants: array [uid1, uid2]
├── lastMessage: string
├── lastMessageSenderId: string
├── lastMessageTime: timestamp
└── createdAt: timestamp

messages/{messageId}
├── messageId: string
├── chatId: string
├── senderId: string
├── text: string
├── createdAt: timestamp
├── read: boolean
├── readAt: timestamp (nullable)
├── deleted: boolean
├── deletedForMe: map {uid: boolean}
├── deletedForAll: boolean
├── edited: boolean
├── editedAt: timestamp (nullable)
└── imageUrl: string (nullable)

blocks/{blockId} (format: blockerId_blockedId)
├── blockerId: string
├── blockedId: string
└── createdAt: timestamp

╔════════════════════════════════════════════════════════════════╗
║                  УРОВНИ ДОСТУПА                                ║
╚════════════════════════════════════════════════════════════════╝

ПОЛЬЗОВАТЕЛИ (users):
✓ Читать: Все авторизованные пользователи
✓ Писать: Только свой профиль

ЧАТЫ (chats):
✓ Читать: Только участники чата
✓ Писать: Только участники (создание/обновление)
✗ Удалять: Запрещено (используем мягкое удаление)

СООБЩЕНИЯ (messages):
✓ Читать: Участники чата, в котором сообщение
✓ Создавать: Участники чата (своего отправителя)
✓ Обновлять: Только автор сообщения
✓ Удалять: Только автор сообщения

БЛОКИРОВКИ (blocks):
✓ Читать: Участники блокировки
✓ Создавать: От себя
✓ Удалять: Только блокирующий

╔════════════════════════════════════════════════════════════════╗
║                  ИНДЕКСЫ FIRESTORE                             ║
╚════════════════════════════════════════════════════════════════╝

Для оптимальной производительности создайте индексы:

1. messages
   - Composite Index: (chatId, createdAt) для быстрой загрузки

2. users
   - Single Field: userId (для поиска по ID)

3. chats
   - Composite Index: (participants, lastMessageTime)

Firestore автоматически предложит создать индексы при первом запросе.
